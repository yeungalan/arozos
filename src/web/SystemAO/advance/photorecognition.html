Using it from AGI
requirelib("imagerecognition");

if (imagerecognition.ready()){
    // Descriptive + object tags
    var tags = imagerecognition.tag("user:/Photo/holiday.jpg");

    // Group the same person across photos under a stable UUID
    var faces = imagerecognition.recognizeFaces("user:/Photo/holiday.jpg");
    for (var i = 0; i < faces.length; i++){
        console.log("person " + faces[i].personUUID + " at " + JSON.stringify(faces[i].box));
    }

    sendJSONResp(JSON.stringify({tags: tags, faces: faces}));
}
